cmake_minimum_required(VERSION 3.16)
project(material-dark-kde VERSION 1.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

option(BUILD_STYLE       "Application style (Qt widget style)" ON)
option(BUILD_DECORATION  "Window decoration (KDecoration3)"    ON)
option(BUILD_PREVIEW     "Alat render galeri widget ke PNG"    OFF)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

if(BUILD_DECORATION)
    find_package(ECM REQUIRED NO_MODULE)
    set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
    include(KDEInstallDirs6)
    include(KDECMakeSettings)
    find_package(KF6CoreAddons REQUIRED)
    find_package(KDecoration3 REQUIRED)
    set(MATERIAL_QT_PLUGIN_DIR ${KDE_INSTALL_PLUGINDIR})
    set(MATERIAL_DATA_DIR ${KDE_INSTALL_DATADIR})
    set(MATERIAL_QML_DIR ${KDE_INSTALL_QMLDIR})
else()
    include(GNUInstallDirs)
    set(MATERIAL_QT_PLUGIN_DIR ${CMAKE_INSTALL_LIBDIR}/qt6/plugins)
    set(MATERIAL_DATA_DIR ${CMAKE_INSTALL_DATADIR})
    set(MATERIAL_QML_DIR ${CMAKE_INSTALL_LIBDIR}/qt6/qml)
endif()

# Style Qt Quick Controls 2 (Switch, ComboBox, Button) untuk halaman QML seperti System Settings
install(DIRECTORY qml/MaterialDark DESTINATION ${MATERIAL_QML_DIR})

if(BUILD_STYLE)
    add_subdirectory(style)
endif()
if(BUILD_DECORATION)
    add_subdirectory(decoration)
endif()
if(BUILD_PREVIEW)
    add_executable(material-preview tools/preview.cpp style/materialstyle.cpp style/materialstyle.h)
    target_link_libraries(material-preview Qt6::Widgets)
endif()
